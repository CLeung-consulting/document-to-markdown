server.py (Flask + MarkItDown API) Python 3.10+ 
from flask import Flask, request, jsonify, render_template_string
from markitdown import MarkItDown
import os
import tempfile
import tiktoken  # OpenAI tiktoken or custom local estimator

app = Flask(__name__)
md_converter = MarkItDown()

@app.route('/api/convert', methods=['POST'])
def convert_document():
    """
    Endpoint to receive uploaded documents (PDF, DOCX, XLSX, PPTX)
    and return token-optimized Markdown.
    """
    if 'file' not in request.files:
        return jsonify({'error': 'No file part in request'}), 400
        
    file = request.files['file']
    if file.filename == '':
        return jsonify({'error': 'No file selected'}), 400

    # Save to temporary file for MarkItDown processing
    filename = file.filename
    ext = os.path.splitext(filename)[1].lower()
    
    with tempfile.NamedTemporaryFile(delete=False, suffix=ext) as tmp:
        file.save(tmp.name)
        tmp_path = tmp.name

    try:
        # Convert using Microsoft MarkItDown
        result = md_converter.convert(tmp_path)
        markdown_text = result.text_content

        # Estimate tokens using tiktoken (cl100k_base / llama style)
        try:
            encoding = tiktoken.get_encoding("cl100k_base")
            token_count = len(encoding.encode(markdown_text))
        except Exception:
            # Fallback estimation (~4 chars per token)
            token_count = len(markdown_text) // 4

        return jsonify({
            'success': True,
            'filename': filename,
            'markdown': markdown_text,
            'token_count': token_count,
            'char_count': len(markdown_text)
        })

    except Exception as e:
        return jsonify({'error': str(e)}), 500
    finally:
        if os.path.exists(tmp_path):
            os.remove(tmp_path)

if __name__ == '__main__':
    print("🚀 DocToMark Python Backend running on http://localhost:5000")
    app.run(host='0.0.0.0', port=5000, debug=True)
